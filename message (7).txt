-- ─────────────────────────────────────────────
--  TWEEN HELPERS
-- ─────────────────────────────────────────────
local function Tween(obj, props, duration, style, direction)
    style     = style     or Enum.EasingStyle.Quart
    direction = direction or Enum.EasingDirection.Out
    local info = TweenInfo.new(duration or 0.25, style, direction)
    local t    = TweenService:Create(obj, info, props)
    t:Play()
    return t
end

local function SpringTween(obj, props, duration)
    return Tween(obj, props, duration or 0.35, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
end

-- ─────────────────────────────────────────────
--  UI FACTORY HELPERS
-- ─────────────────────────────────────────────
local function Make(class, props)
    local obj = Instance.new(class)
    for k, v in pairs(props) do
        if k ~= "Parent" then
            obj[k] = v
        end
    end
    if props.Parent then obj.Parent = props.Parent end
    return obj
end

local function AddCorner(parent, radius)
    Make("UICorner", { CornerRadius = UDim.new(0, radius or 8), Parent = parent })
end

local function AddStroke(parent, color, thickness, transparency)
    Make("UIStroke", {
        Color        = color or Theme.Border,
        Thickness    = thickness or 1,
        Transparency = transparency or Theme.BorderTransp,
        Parent       = parent,
    })
end

local function AddPadding(parent, top, right, bottom, left)
    Make("UIPadding", {
        PaddingTop    = UDim.new(0, top    or 8),
        PaddingRight  = UDim.new(0, right  or 8),
        PaddingBottom = UDim.new(0, bottom or 8),
        PaddingLeft   = UDim.new(0, left   or 8),
        Parent        = parent,
    })
end

local function AddListLayout(parent, direction, spacing, align)
    Make("UIListLayout", {
        FillDirection       = direction or Enum.FillDirection.Vertical,
        SortOrder           = Enum.SortOrder.LayoutOrder,
        Padding             = UDim.new(0, spacing or 6),
        HorizontalAlignment = align or Enum.HorizontalAlignment.Left,
        Parent              = parent,
    })
end

local function Label(text, parent, size, color, weight, xAlign)
    return Make("TextLabel", {
        Text               = text,
        Font               = weight == "bold" and Enum.Font.GothamBold or Enum.Font.Gotham,
        TextSize           = size or 12,
        TextColor3         = color or Theme.TextPrimary,
        BackgroundTransparency = 1,
        TextXAlignment     = xAlign or Enum.TextXAlignment.Left,
        Size               = UDim2.new(1, 0, 0, size and size + 6 or 18),
        Parent             = parent,
    })
end

--  DRAGGING
-- ─────────────────────────────────────────────
do
    local dragging, dragStart, startPos = false, nil, nil

    TitleBar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging  = true
            dragStart = input.Position
            startPos  = Window.Position
        end
    end)

    TitleBar.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
            local delta = input.Position - dragStart
            Window.Position = UDim2.new(
                startPos.X.Scale,
                startPos.X.Offset + delta.X,
                startPos.Y.Scale,
                startPos.Y.Offset + delta.Y
            )
        end
    end)
end

-- Toggle component: returns the toggle frame, and a function to set state
local function Toggle(parent, xOffset, yOffset, initialState, onChange)
    local state = initialState or false

    local bg = Make("Frame", {
        Size             = UDim2.new(0, 40, 0, 21),
        Position         = UDim2.new(1, -(xOffset or 0) - 40, 0.5, -10),
        BackgroundColor3 = state and Theme.ToggleOn or Theme.ToggleOff,
        BorderSizePixel  = 0,
        ZIndex           = 4,
        Parent           = parent,
    })
    AddCorner(bg, 99)
    AddStroke(bg, state and Theme.Cyan or Theme.Border, 1, state and 0.5 or 0.7)

    local knob = Make("Frame", {
        Size             = UDim2.new(0, 15, 0, 15),
        Position         = state and UDim2.new(1, -18, 0.5, -7) or UDim2.new(0, 3, 0.5, -7),
        BackgroundColor3 = state and Theme.ToggleKnobOn or Theme.ToggleKnobOff,
        BorderSizePixel  = 0,
        ZIndex           = 5,
        Parent           = bg,
    })
    AddCorner(knob, 99)

    local btn = Make("TextButton", {
        Text             = "",
        Size             = UDim2.new(1, 0, 1, 0),
        BackgroundTransparency = 1,
        ZIndex           = 6,
        Parent           = bg,
    })

    btn.MouseButton1Click:Connect(function()
        state = not state
        Tween(bg,   { BackgroundColor3 = state and Theme.ToggleOn or Theme.ToggleOff }, 0.2)
        Tween(knob, {
            Position         = state and UDim2.new(1,-18,0.5,-7) or UDim2.new(0,3,0.5,-7),
            BackgroundColor3 = state and Theme.ToggleKnobOn or Theme.ToggleKnobOff,
        }, 0.25, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
        local stroke = bg:FindFirstChildOfClass("UIStroke")
        if stroke then stroke.Color = state and Theme.Cyan or Theme.Border end
        if onChange then onChange(state) end
    end)

    return bg, function() return state end
end

        ZIndex           = 5,

    })

        Tween(row, {BackgroundTransparency = 1}, 0.2)
        Tween(row, {Size = UDim2.new(1,-4,0,0)}, 0.2)
        task.delay(0.2, function()
            row:Destroy()

end

local function AddUsername(name)
    name = name:match("^%s*(.-)%s*$"):gsub("^@", "")
    if name == "" then return end
    if table.find(State.Usernames, name) then
        -- Flash red to indicate duplicate
        Tween(UnameInput:FindFirstChildOfClass("UIStroke"), {Color = Theme.Danger}, 0.15)
        task.delay(0.5, function()
            Tween(UnameInput:FindFirstChildOfClass("UIStroke"), {Color = Theme.Cyan}, 0.3)
        end)
        return
    end
    table.insert(State.Usernames, name)
    RenderUsernameItem(name, #State.Usernames)
    UpdateCount()
    UnameInput.Text = ""
end

-- ─────────────────────────────────────────────
--  TAB BUTTON BUILDER  (runs after panels exist)
-- ─────────────────────────────────────────────
local TabButtons = {}

local TabDefs = {
    { id = "usernames", icon = "👤", label = "Usernames", panel = UsernamesPanel, order = 1 },
    { id = "settings",  icon = "⚙",  label = "Settings",  panel = SettingsPanel,  order = 2 },
}

local activeTab = nil

local function SwitchTab(def)
    if activeTab == def.id then return end
    activeTab = def.id
    State.ActiveTab = def.id

    for _, td in pairs(TabDefs) do
        local btn = TabButtons[td.id]
        if btn then
            if td.id == def.id then
                Tween(btn, {
                    BackgroundTransparency = 0.88,
                    TextColor3 = Theme.Cyan,
                }, 0.2)
                btn:FindFirstChildOfClass("UIStroke").Color = Theme.Cyan
            else
                Tween(btn, {
                    BackgroundTransparency = 1,
                    TextColor3 = Theme.TextMuted,
                }, 0.2)
                btn:FindFirstChildOfClass("UIStroke").Color = Theme.Border
            end
        end
        -- Animate panel
        if td.id == def.id then
            td.panel.Visible = true
            td.panel.Position = UDim2.new(0.05, 0, 0, 0)
            Tween(td.panel, {Position = UDim2.new(0,0,0,0)}, 0.25)
        else
            Tween(td.panel, {Position = UDim2.new(-0.05,0,0,0)}, 0.2)
            task.delay(0.2, function()
                if activeTab ~= td.id then
                    td.panel.Visible = false
                end
            end)
        end
    end
end

for _, def in ipairs(TabDefs) do
    local isFirst = def.order == 1
    local tabBtn = Make("TextButton", {
        Name             = "Tab_" .. def.id,
        Text             = def.icon .. "  " .. def.label:upper(),
        Font             = Enum.Font.GothamBold,
        TextSize         = 10,
        TextColor3       = isFirst and Theme.Cyan or Theme.TextMuted,
        BackgroundColor3 = Theme.Cyan,
        BackgroundTransparency = isFirst and 0.88 or 1,
        BorderSizePixel  = 0,
        Size             = UDim2.new(0, 120, 0, 28),
        LayoutOrder      = def.order,
        ZIndex           = 4,
        Parent           = TabBar,
    })
    AddCorner(tabBtn, 7)
    AddStroke(tabBtn, isFirst and Theme.Cyan or Theme.Border, 1, isFirst and 0.6 or 0.85)

    TabButtons[def.id] = tabBtn
    if isFirst then
        def.panel.Visible = true
        activeTab = def.id
    end

    tabBtn.MouseButton1Click:Connect(function()
        SwitchTab(def)
    end)
    tabBtn.Activated:Connect(function()  -- also respond to :Activated for programmatic switching
        SwitchTab(def)
    end)
end
